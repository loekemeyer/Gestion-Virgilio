const WA_PHONE_ID_FALLBACK = "918089688061759";
const DESTINATARIOS = ["5491162521635", "5491131181594"];

const SUPABASE_URL = "https://hrxfctzncixxqmpfhskv.supabase.co";
const SUPABASE_SERVICE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || "";

const CORS_HEADERS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

async function leerSecret(sb: any, clave: string, fallback?: string): Promise<string> {
  try {
    const { data, error } = await sb.rpc("server_secret", { p_k: clave });
    if (!error && data) return String(data);
    if (error) console.error(`server_secret(${clave}): ${error.message}`);
  } catch (err) {
    console.error(`server_secret(${clave}): ${err}`);
  }
  if (fallback !== undefined) return fallback;
  throw new Error(`falta el secreto ${clave} en lecturacvs.server_secrets`);
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { headers: CORS_HEADERS });
  }

  try {
    const sb = createClient(SUPABASE_URL, SUPABASE_SERVICE_KEY);
    const WA_TOKEN = await leerSecret(sb, "REPORTES_WA_TOKEN");
    const WA_PHONE_ID = await leerSecret(sb, "REPORTES_WA_PHONE_ID", WA_PHONE_ID_FALLBACK);

    const { data: matrices } = await sb.from("Matrices").select("N_Matriz, Matriz, Tiempo_Historico");
    if (!matrices) throw new Error("No se pudieron cargar matrices");

    const allRegs: any[] = [];
    const PAGE = 1000;
    let from = 0;
    while (true) {
      const { data } = await sb.from("db_n8n_espejo")
        .select("Matriz, Uni, Segundos_Trabajados, Anular_Tiempo, Eliminar, Premio, Tiempo_Toma, Legajo")
        .or("Eliminar.is.null,Eliminar.neq.S")
        .neq("Legajo", "1")
        .range(from, from + PAGE - 1);
      if (!data || !data.length) break;
      allRegs.push(...data);
      if (data.length < PAGE) break;
      from += PAGE;
    }

    const stats = matrices
      .filter(m => Number(m.Tiempo_Historico || 0) > 0)
      .map(m => {
        const nMat = String(m.N_Matriz || "").trim();
        const tHist = Number(m.Tiempo_Historico || 0);
        const regs = allRegs.filter(r => String(r.Matriz || "").trim() === nMat);
        const active = regs.filter(r => !r.Anular_Tiempo);

        if (active.length === 0) return null;

        const tiempos = active.map(r => {
          if (r.Tiempo_Toma != null && Number(r.Tiempo_Toma) > 0) return Number(r.Tiempo_Toma);
          const uni = Number(r.Uni || 1) || 1;
          return Number(r.Segundos_Trabajados || 0) / uni;
        });
        const prom = tiempos.reduce((a, b) => a + b, 0) / tiempos.length;
        const diff = prom - tHist;
        const premios = active.map(r => Number(r.Premio || 0));
        const ptje = premios.reduce((a, b) => a + b, 0) / premios.length;

        return {
          num: nMat,
          nombre: m.Matriz || "",
          tHist,
          prom: prom.toFixed(1),
          diff: diff.toFixed(1),
          ptje: ptje.toFixed(2),
          regs: active.length
        };
      })
      .filter(Boolean)
      .sort((a: any, b: any) => Number(a.num) - Number(b.num));

    const now = new Date();
    const argTime = now.toLocaleString("es-AR", { timeZone: "America/Argentina/Buenos_Aires" });

    let msg = `📊 *Rendimiento Matrices*\n📅 ${argTime}\n\n`;
    msg += `N° | Nombre | T.Hist | Prom | Diff | Ptje\n`;
    msg += `─────────────────────────\n`;

    for (const s of stats as any[]) {
      const diffSign = Number(s.diff) > 0 ? "+" : "";
      const ptjeSign = Number(s.ptje) > 0 ? "+" : "";
      msg += `${s.num} | ${s.nombre.substring(0, 20)} | ${s.tHist} | ${s.prom} | ${diffSign}${s.diff} | ${ptjeSign}${s.ptje}\n`;
    }

    msg += `\n📋 Total: ${stats.length} matrices con tiempo`;

    const url = `https://graph.facebook.com/v21.0/${WA_PHONE_ID}/messages`;
    const resultados: { numero: string; ok: boolean; error?: string }[] = [];

    for (const num of DESTINATARIOS) {
      try {
        const res = await fetch(url, {
          method: "POST",
          headers: {
            Authorization: `Bearer ${WA_TOKEN}`,
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            messaging_product: "whatsapp",
            to: num,
            type: "text",
            text: { body: msg },
          }),
        });
        const data = await res.json();
        resultados.push({ numero: num, ok: res.ok, error: res.ok ? undefined : data?.error?.message });
      } catch (err) {
        resultados.push({ numero: num, ok: false, error: String(err) });
      }
    }

    const enviados = resultados.filter(r => r.ok).length;
    return new Response(
      JSON.stringify({ enviados, total: DESTINATARIOS.length, stats: stats.length, resultados }),
      { headers: { ...CORS_HEADERS, "Content-Type": "application/json" } }
    );
  } catch (err) {
    return new Response(
      JSON.stringify({ error: String(err) }),
      { status: 500, headers: { ...CORS_HEADERS, "Content-Type": "application/json" } }
    );
  }
});
